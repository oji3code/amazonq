cat <<EOF | kubectl apply -f -
apiVersion: v1

  name: karpenter-node-logger
EOF

#==========================================================
# Create SA, Role, ClusterRoleBinding
#==========================================================

cat <<EOF | kubectl apply -f -
apiVersion: v1
kind: ServiceAccount
metadata:
  name: karpenter-node-logger
  namespace: karpenter-node-logger
EOF

cat <<EOF | kubectl apply -f -
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRole
metadata:
  name: karpenter-node-logger # 任意でわかりやすい名前に変更可能
rules:
- apiGroups: [""] # "" はコアAPIグループ（ノードなど）を示す
  resources: ["nodes"]
  verbs: ["get", "list", "watch"] # get だけでなく、list と watch もあると良い
- apiGroups: ["karpenter.sh"] # NodeClaimリソースのAPIグループ
  resources: ["nodeclaims"] # 監視したいリソース
  verbs: ["get", "list", "watch"] # 取得、一覧表示、監視の権限を付与
EOF

cat <<EOF | kubectl apply -f -
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRoleBinding
metadata:
  name: karpenter-node-logger-binding # 任意でわかりやすい名前に変更可能
subjects:
- kind: ServiceAccount
  name: karpenter-node-logger # エラーメッセージに表示されたサービスアカウント名
  namespace: karpenter-node-logger # エラーメッセージに表示されたサービスアカウントの名前空間
roleRef:
  kind: ClusterRole
  name: karpenter-node-logger # 上記で作成したClusterRoleの名前
  apiGroup: rbac.authorization.k8s.io
EOF

#==========================================================
# Create Deployment(Python App: Monitor for Kubernetes's Event of Node)
#==========================================================

cat <<EOF | kubectl apply -f -
apiVersion: apps/v1
kind: Deployment
metadata:
  name: karpenter-node-logger
  namespace: karpenter-node-logger
spec:
  replicas: 1
  selector:
    matchLabels:
      app: karpenter-node-logger
  template:
    metadata:
      labels:
        app: karpenter-node-logger
      annotations:
        ad.datadoghq.com/karpenter-node-logger.logs: |-
          [
            {
              "source": "python",
              "service": "karpenter"
            }
          ]
    spec:
      serviceAccountName: karpenter-node-logger
      containers:
      - name: karpenter-node-logger
        image: python:3.9-slim
        command:
        - /bin/bash
        - -c
        - |
          pip install kubernetes requests > /dev/null 2>&1
          mkdir /app
          cat > /app/karpenter-node-logger.py << 'EOF_PYTHON'
          from kubernetes import client, config, watch
          import logging
          # DEBUGレベルのログもDatadogに送りたい場合は logging.DEBUG に変更
          logging.basicConfig(level=logging.INFO,
                              format='%(message)s') # タイムスタンプを削除

          logger = logging.getLogger('karpenter-node-logger')

          def logging_karpenter_node_scale():
            try:
              config.load_kube_config()
            except config.ConfigException:
              # Pod内で実行する場合は in-cluster config にフォールバック
              logger.info("Falling back to in-cluster config...")
              config.load_incluster_config()

            core_api = client.CoreV1Api()
            w = watch.Watch()
            custom_api = client.CustomObjectsApi()

            logger.debug("Starting to watch NodeClaims for scale events...")

            for event in w.stream(custom_api.list_cluster_custom_object,
                                    group="karpenter.sh",
                                    version="v1",
                                    plural="nodeclaims"):
              
              nodeclaim = event['object']
              event_type = event['type']
              nodeclaim_name = nodeclaim['metadata']['name']
              node_name = nodeclaim.get('status', {}).get('nodeName')
              
              instance_id = None
              provider_id = nodeclaim.get('status', {}).get('providerID')
              if provider_id:
                instance_id = provider_id.split('/')[-1]

              node_ip_address = None
              if node_name:
                try:
                  node_obj = core_api.read_node(name=node_name)
                  for address in node_obj.status.addresses:
                    if address.type == 'InternalIP':
                      node_ip_address = address.address
                      break
                except client.ApiException as e:
                  logger.warning(f"Warning: Could not get Node details for {node_name}: {e}")

              # Event発生事由が「起動中」or「停止処理中」であるか
              current_nodeclaim_reason = None
              is_ready = False
              for condition in nodeclaim.get('status', {}).get('conditions', []):
                if condition.get('type') == 'Ready' and condition.get('status') == 'True':
                  is_ready = True
                if condition.get('type') == 'Disrupting' and condition.get('status') == 'True':
                  current_nodeclaim_reason = "Disrupting"

              # ログメッセージのベースを構築
              base_log_message = f"NodeClaim: {nodeclaim_name}, EventType: {event_type}"
              if node_name:
                base_log_message += f", NodeName: {node_name}"
              if instance_id:
                base_log_message += f", InstanceID: {instance_id}"
              if node_ip_address:
                base_log_message += f", IP: {node_ip_address}"

              # --- スケールイベント(アウト・イン)の検知ロジック ---
              if event_type == 'ADDED':
                logger.info(f"+++ DETECTED SCALE-OUT (NodeClaim ADDED): {base_log_message} +++")
                logger.debug(f"  Status: NodeClaim created. Waiting for node to be ready...")
              elif event_type == 'MODIFIED':
                if is_ready and not nodeclaim.get('metadata', {}).get('deletionTimestamp'):
                  logger.debug(f"+++ DETECTED SCALE-OUT (Node READY): {base_log_message} +++")
                  logger.debug(f"  Status: Node is fully ready and joined the cluster.")
                elif current_nodeclaim_reason == "Disrupting":
                  logger.debug(f"--- DETECTED SCALE-IN (NodeClaim DISRUPTING): {base_log_message} ---")
                  logger.debug(f"  Status: Node is being disrupted/terminated.")
              elif event_type == 'DELETED':
                logger.info(f"--- DETECTED SCALE-IN (NodeClaim DELETED): {base_log_message} ---")
                logger.debug(f"  Status: NodeClaim deleted. Node has been terminated.")
              else:
                logger.debug(f"--- Other NodeClaim Event: {base_log_message} ---")
                logger.debug(f"  Status: Current reason - {current_nodeclaim_reason}, Ready: {is_ready}")
                  
          if __name__ == "__main__":
            logging_karpenter_node_scale()
          EOF_PYTHON
          
          python /app/karpenter-node-logger.py
        resources:
          requests:
            memory: "128Mi"
            cpu: "100m"
          limits:
            memory: "256Mi"
            cpu: "200m"
EOF
